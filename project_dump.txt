===== ./README.md =====
# Family Points

A PHP + MySQL family rewards app for tracking children’s points, managing rewards, and supporting family-scoped access.

## Current user flow

See the project flow document here:

- `docs/current-user-flow.md`

## Core concept

This app is designed around a family-first model:

- Parents log in to manage their household
- Children access their own page through a unique access code/link
- Family data is isolated by `family_id`
- The creator can use developer-only family switching for testing and debugging

## Key pages

- `landing.php` – public landing page, no household data shown
- `parent-login.php` – parent login
- `parent-dashboard.php` – family management dashboard
- `index.php` – family view for logged-in parents
- `shop.php` – child-specific rewards page, protected by valid access code or parent login
- `family-settings.php` – family settings page
- `creator-admin.php` – creator-only maintenance tools

## Security model

- Family data is protected by `family_id`
- Child access uses a per-child access code/link rather than per-child login accounts
- Parent pages require valid parent session
- Creator-only tools remain separate from normal family access

## Local setup

1. Start MAMP / Apache / MySQL.
2. Import the database schema if needed.
3. Confirm the database configuration in:
   - `includes/config.php`
   - `includes/config.local.php`
4. Open the project in the browser using the local MAMP URL.

## Development notes

- The app is PHP procedural code with PDO and MySQL.
- The current architecture favors family boundaries before deeper multi-parent role complexity.
- Public pages should not expose any family-specific data.

## Related docs

- `docs/current-user-flow.md`
- `docs/multi-family-migration-plan.md`
- `docs/creator-permissions.md`
- `docs/creator-admin-ops.md`

===== ./approve-reward.php =====
<?php

require_once __DIR__ . '/includes/auth.php';
require_once __DIR__ . '/includes/db.php';

if ($_SERVER['REQUEST_METHOD'] !== 'POST') {
    header('Location: parent-rewards.php');
    exit;
}

verifyCsrf();

$redemptionId = (int)($_POST['redemption_id'] ?? 0);
$action = $_POST['action'] ?? '';
$familyId = currentParentFamilyId();

if ($familyId === null || ! in_array($action, ['approve', 'reject'], true)) {
    http_response_code(403);
    exit('无权执行此操作。');
}

$pdo = db();
$pdo->beginTransaction();

try {
    $stmt = $pdo->prepare(
        'SELECT rd.*, r.cost, r.title, c.id AS child_id, c.normal_balance, c.theme, c.family_id
         FROM redemptions rd
         JOIN rewards r ON r.id = rd.reward_id
         JOIN children c ON c.id = rd.child_id
         WHERE rd.id = ? AND rd.status = "pending" AND c.family_id = ?
         FOR UPDATE'
    );
    $stmt->execute([$redemptionId, $familyId]);
    $redemption = $stmt->fetch();

    if (! $redemption) {
        throw new Exception('兑换请求不存在，或不属于当前家庭。');
    }

    if ($action === 'reject') {
        $update = $pdo->prepare(
            'UPDATE redemptions
             SET status = "rejected", approved_at = NOW()
             WHERE id = ?'
        );
        $update->execute([$redemptionId]);

        $pdo->commit();
        header('Location: parent-rewards.php');
        exit;
    }

    if ($redemption['normal_balance'] < $redemption['cost']) {
        throw new Exception('积分不足，无法确认兑换。');
    }

    $deduct = $pdo->prepare(
        'UPDATE children
         SET normal_balance = normal_balance - ?
         WHERE id = ? AND normal_balance >= ? AND family_id = ?'
    );
    $deduct->execute([
        $redemption['cost'],
        $redemption['child_id'],
        $redemption['cost'],
        $familyId
    ]);

    if ($deduct->rowCount() !== 1) {
        throw new Exception('积分不足，无法确认兑换。');
    }

    $approve = $pdo->prepare(
        'UPDATE redemptions
         SET status = "approved", approved_at = NOW()
         WHERE id = ?'
    );
    $approve->execute([$redemptionId]);

    $pdo->commit();

    header('Location: parent-rewards.php');
    exit;
} catch (Exception $e) {
    $pdo->rollBack();
    exit(htmlspecialchars($e->getMessage()));
}

<?php
require_once __DIR__ . '/includes/auth.php';
require_once __DIR__ . '/includes/db.php';

$familyId = currentParentFamilyId();
if ($familyId === null) {
    header('Location: parent-login.php');
    exit;
}

$notice = '';

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    verifyCsrf();

    $action = $_POST['action'] ?? '';

    if ($action === 'add_child') {
        $childName = trim((string)($_POST['child_name'] ?? ''));
        $theme = in_array($_POST['theme'] ?? '', ['minecraft', 'music'], true)
            ? $_POST['theme']
            : 'minecraft';

        if ($childName === '') {
            $notice = '孩子姓名不能为空。';
        } else {
            $stmt = db()->prepare(
                'INSERT INTO children (name, theme, family_id, normal_balance, daily_points_today, last_score_date, quarterly_pool, daily_cap, quarterly_goal, created_at)
                 VALUES (?, ?, ?, 0, 0, NULL, 0, 25, 150, NOW())'
            );
            $stmt->execute([$childName, $theme, $familyId]);
            $notice = '已添加孩子：' . $childName;
        }
    }

    if ($action === 'update_child') {
        $childId = (int)($_POST['child_id'] ?? 0);
        $childName = trim((string)($_POST['child_name'] ?? ''));
        $theme = in_array($_POST['theme'] ?? '', ['minecraft', 'music'], true)
            ? $_POST['theme']
            : 'minecraft';

        if ($childId <= 0 || $childName === '') {
            $notice = '孩子信息不完整。';
        } else {
            $check = db()->prepare('SELECT 1 FROM children WHERE id = ? AND family_id = ?');
            $check->execute([$childId, $familyId]);

            if (! $check->fetch()) {
                http_response_code(403);
                exit('无权修改此孩子。');
            }

            $stmt = db()->prepare('UPDATE children SET name = ?, theme = ? WHERE id = ? AND family_id = ?');
            $stmt->execute([$childName, $theme, $childId, $familyId]);
            $notice = '已更新孩子信息。';
        }
    }

    if ($action === 'generate_child_access_code') {
        $childId = (int)($_POST['child_id'] ?? 0);
        if ($childId <= 0) {
            $notice = '无效的孩子记录。';
        } else {
            $check = db()->prepare('SELECT 1 FROM children WHERE id = ? AND family_id = ?');
            $check->execute([$childId, $familyId]);
            if (! $check->fetch()) {
                http_response_code(403);
                exit('无权管理此孩子。');
            }

            $code = issueChildAccessCode($childId);
            $notice = '已生成新访问码：' . $code;
        }
    }

    if ($action === 'delete_child') {
        $childId = (int)($_POST['child_id'] ?? 0);

        if ($childId <= 0) {
            $notice = '无效的孩子记录。';
        } else {
            $check = db()->prepare('SELECT 1 FROM children WHERE id = ? AND family_id = ?');
            $check->execute([$childId, $familyId]);
            if (! $check->fetch()) {
                http_response_code(403);
                exit('无权删除此孩子。');
            }

            $pending = db()->prepare('SELECT COUNT(*) FROM redemptions WHERE child_id = ? AND status = "pending"');
            $pending->execute([$childId]);
            if ((int)$pending->fetchColumn() > 0) {
                $notice = '请先处理此孩子的待确认奖励。';
            } else {
                $events = db()->prepare('SELECT COUNT(*) FROM point_events WHERE child_id = ?');
                $events->execute([$childId]);
                if ((int)$events->fetchColumn() > 0) {
                    $notice = '此孩子已有积分记录，不能直接删除。';
                } else {
                    $delete = db()->prepare('DELETE FROM children WHERE id = ? AND family_id = ?');
                    $delete->execute([$childId, $familyId]);
                    $notice = '已删除孩子记录。';
                }
            }
        }
    }

    if ($action === 'switch_dev_family') {
        if (! isAppCreator()) {
            http_response_code(403);
            exit('此操作仅限应用创建者。');
        }

        $targetFamilyId = (int)($_POST['dev_family_id'] ?? 0);
        $familyCheck = db()->prepare('SELECT id FROM families WHERE id = ?');
        $familyCheck->execute([$targetFamilyId]);

        if (! $familyCheck->fetch()) {
            http_response_code(403);
            exit('无效的家庭视角。');
        }

        $_SESSION['dev_family_id'] = $targetFamilyId;
        $redirectTo = trim((string)($_POST['redirect_to'] ?? 'child-settings.php'));
        $safeRedirect = in_array($redirectTo, ['index.php', 'parent-dashboard.php', 'parent-record.php', 'parent-rewards.php', 'child-settings.php', 'point-reward-settings.php', 'family-settings.php', 'shop.php'], true)
            ? $redirectTo
            : 'child-settings.php';
        header('Location: ' . $safeRedirect);
        exit;
    }
}

$familyStmt = db()->prepare('SELECT name FROM families WHERE id = ?');
$familyStmt->execute([$familyId]);
$family = $familyStmt->fetch();
$familyName = $family['name'] ?? 'Default Family';

$childrenStmt = db()->prepare('SELECT * FROM children WHERE family_id = ? ORDER BY id');
$childrenStmt->execute([$familyId]);
$children = $childrenStmt->fetchAll();

    header('Location: index.php');
    exit;

$familyId = currentParentFamilyId();
if ($familyId === null) {
    http_response_code(403);
    exit('无权访问此功能。');
}

$pdo = db();
$pdo->beginTransaction();

try {
    $rewardStmt = $pdo->prepare(
        'SELECT r.*
         FROM rewards r
         JOIN children c ON c.id = r.child_id
         WHERE r.id = ? AND r.is_active = 1 AND c.family_id = ?
         FOR UPDATE'
    );
    $rewardStmt->execute([$rewardId, $familyId]);
    $reward = $rewardStmt->fetch();

    if (! $reward) {
        throw new Exception('奖励不存在或不属于当前家庭。');
    }

    $childStmt = $pdo->prepare(
        'SELECT * FROM children WHERE id = ? AND family_id = ? FOR UPDATE'
    );
    $childStmt->execute([$reward['child_id'], $familyId]);
    $child = $childStmt->fetch();

    if (! $child) {
        throw new Exception('孩子不存在或不属于当前家庭。');
    }

    $pendingStmt = $pdo->prepare(
        'SELECT COUNT(*) FROM redemptions
         WHERE child_id = ? AND status = "pending"'
    );
    $pendingStmt->execute([$child['id']]);

    if ($pendingStmt->fetchColumn() > 0) {
        $shop = $child['theme'] === 'minecraft'
            ? 'shop-minecraft.php?error=duplicate'
            : 'shop-music.php?error=duplicate';

        header('Location: ' . $shop);
        exit;
    }

    if ($child['normal_balance'] < $reward['cost']) {
        throw new Exception('积分不足。');
    }

    $insert = $pdo->prepare(
        'INSERT INTO redemptions (child_id, reward_id, cost, status)
         VALUES (?, ?, ?, "pending")'
    );
    $insert->execute([
        $child['id'],
        $reward['id'],
        $reward['cost']
    ]);

    $pdo->commit();

    header('Location: ' . ($child['theme'] === 'minecraft'
    ? 'shop-minecraft.php?requested=1'
    : 'shop-music.php?requested=1'));
    exit;
} catch (Exception $e) {
    $pdo->rollBack();
    exit(htmlspecialchars($e->getMessage()));
}

-- Multi-family migration for this app
-- Current schema in local MAMP DB:
-- - parent_users(id, username, password_hash, created_at)
-- - children(id, name, theme, normal_balance, daily_points_today, last_score_date,
--            quarterly_pool, daily_cap, quarterly_goal, created_at)
--
-- Safe plan:
-- 1. Backup DB
-- 2. Add families table
-- 3. Add family_id columns to parent_users and children
-- 4. Backfill with a default family
-- 5. Validate data
-- 6. Add foreign keys only if data is verified

-- -----------------------------------------------------------------------------
-- 1) Create family table
-- -----------------------------------------------------------------------------
CREATE TABLE IF NOT EXISTS families (
    id INT UNSIGNED NOT NULL AUTO_INCREMENT,
    name VARCHAR(255) NOT NULL,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (id),
    UNIQUE KEY uq_families_name (name)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;

-- -----------------------------------------------------------------------------
-- 2) Add family_id to parent and child records
-- -----------------------------------------------------------------------------
ALTER TABLE parent_users
    ADD COLUMN family_id INT UNSIGNED NULL AFTER id;

ALTER TABLE children
    ADD COLUMN family_id INT UNSIGNED NULL AFTER id;

-- -----------------------------------------------------------------------------
-- 3) Add indexes
-- -----------------------------------------------------------------------------
ALTER TABLE parent_users
    ADD INDEX idx_parent_users_family_id (family_id);

ALTER TABLE children
    ADD INDEX idx_children_family_id (family_id);

-- -----------------------------------------------------------------------------
-- 4) Backfill legacy data into a default family
--    If you already have a real household grouping in your data, replace this with
--    a more precise family assignment process.
-- -----------------------------------------------------------------------------
INSERT INTO families (name)
SELECT 'Default Family'
FROM DUAL
WHERE NOT EXISTS (
    SELECT 1 FROM families WHERE name = 'Default Family'
);

UPDATE parent_users
SET family_id = (
    SELECT id
    FROM families
    WHERE name = 'Default Family'
)
WHERE family_id IS NULL;

UPDATE children
SET family_id = (
    SELECT id
    FROM families
    WHERE name = 'Default Family'
)
WHERE family_id IS NULL;

-- 5) Optional: add foreign keys after verifying all rows are mapped properly.

function currentParent(): ?array
{
    if (empty($_SESSION['parent_id'])) {
        return null;
    }

    $stmt = db()->prepare('SELECT * FROM parent_users WHERE id = ?');
    $stmt->execute([$_SESSION['parent_id']]);
    $user = $stmt->fetch();

    if ($user === false || $user === null) {
        return null;
    }

    return $user;
}

function currentParentFamilyId(): ?int
{
    $user = currentParent();
    if ($user === null) {
        return null;
    }

    if (isAppCreator() && ! empty($_SESSION['dev_family_id'])) {
        $requestedFamilyId = (int) $_SESSION['dev_family_id'];
        if ($requestedFamilyId > 0) {
            $familyCheck = db()->prepare('SELECT id FROM families WHERE id = ?');
            $familyCheck->execute([$requestedFamilyId]);
            if ($familyCheck->fetch()) {
                return $requestedFamilyId;
            }
        }
    }

    $familyId = $user['family_id'] ?? null;
    if ($familyId === null || $familyId === false) {
        return null;
    }

    return (int) $familyId;
}

function appCreatorParentIds(): array
{
    $values = [];

    if (defined('APP_CREATOR_PARENT_IDS') && is_array(APP_CREATOR_PARENT_IDS)) {
        foreach (APP_CREATOR_PARENT_IDS as $id) {
            $value = (int) $id;
            if ($value > 0) {
                $values[] = $value;
            }
        }
    }

    if (defined('APP_CREATOR_PARENT_ID') && (int) APP_CREATOR_PARENT_ID > 0) {
        $values[] = (int) APP_CREATOR_PARENT_ID;
    }

    if (! $values) {
        return [];
    }

    return array_values(array_unique($values));
}

function isAppCreator(?int $parentId = null): bool
{
    $parentId = $parentId ?? (int)($_SESSION['parent_id'] ?? 0);

    if ($parentId <= 0) {
        return false;
    }

    if (in_array($parentId, appCreatorParentIds(), true)) {
        return true;
    }

    try {
        $stmt = db()->prepare('SELECT is_creator FROM parent_users WHERE id = ?');
        $stmt->execute([$parentId]);
        $value = $stmt->fetchColumn();

        if ($value === false || $value === null) {
            return false;
        }

        return (bool) $value;
    } catch (Throwable $e) {
        return false;
    }
}

function requireAppCreator(): void
{
    if (! isAppCreator()) {
        http_response_code(403);
        exit('此操作仅限应用创建者。');
    }
}

function requireParentAccessToChild(int $childId): void
{
    $familyId = currentParentFamilyId();

    if ($familyId === null) {
        http_response_code(403);
        exit('无权访问此数据。');
    }

    $stmt = db()->prepare('SELECT 1 FROM children WHERE id = ? AND family_id = ?');
    $stmt->execute([$childId, $familyId]);

    if (! $stmt->fetch()) {
        http_response_code(403);
        exit('无权访问此数据。');
    }
}

function verifyCsrf(): void
{
    if (! hash_equals($_SESSION['csrf_token'], $_POST['csrf_token'] ?? '')) {
        $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
        http_response_code(403);
        exit('会话已过期，已为您刷新安全令牌。请返回重试。');
    }
}

{
    if (! empty($_SESSION['parent_id'])) {

function calculateAward(int $basePoints, string $completionType): int
{
    $isSelf = str_contains($completionType, 'self_initiated');
    $isExcellent = str_contains($completionType, 'excellent');

    $points = $isSelf ? $basePoints * 2 : $basePoints;

    if ($isExcellent) {
        $points += 1;
    }

    return $points;

function recordPointEvent(
    int $childId,
    ?int $behaviorId,
    string $behaviorLabel,
    int $basePoints,
    string $completionType,
    string $eventDate,
    ?string $note
): array {
    $pdo = db();

    $child = $pdo->prepare('SELECT * FROM children WHERE id = ?');
    $child->execute([$childId]);
    $child = $child->fetch();

    if (! $child) {
        throw new Exception('孩子不存在');
    }

    $today = date('Y-m-d');

    if ($child['last_score_date'] !== $today) {
        $reset = $pdo->prepare(
            'UPDATE children
            SET daily_points_today = 0,
                last_score_date = ?
            WHERE id = ?'
        );
        $reset->execute([$today, $childId]);

        $child['daily_points_today'] = 0;
        $child['last_score_date'] = $today;
    }

    $awarded = calculateAward($basePoints, $completionType);
    $remaining = max(0, $child['daily_cap'] - $child['daily_points_today']);
    $toBalance = min($awarded, $remaining);
    $toPool = $awarded - $toBalance;

    $stmt = $pdo->prepare(
        'INSERT INTO point_events
         (child_id, behavior_id, behavior_label, base_points, awarded_points, overflow_points, completion_type, event_date, note)
         VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)'
    );
    $stmt->execute([
        $childId, $behaviorId, $behaviorLabel, $basePoints,
        $awarded, $toPool, $completionType, $eventDate, $note
    ]);

    $update = $pdo->prepare(
        'UPDATE children
         SET normal_balance = normal_balance + ?,
             daily_points_today = daily_points_today + ?,
             quarterly_pool = quarterly_pool + ?
         WHERE id = ?'
    );
    $update->execute([$toBalance, $awarded, $toPool, $childId]);

    return [
        'awarded' => $awarded,
        'to_balance' => $toBalance,
        'to_pool' => $toPool
    ];
}

# Creator Admin Operations Manual

## Scope

This document defines the operational rules for the application creator account.

The application is intentionally designed so that:

- all normal family accounts are members
- the app owner is the creator / system maintainer
- family-level permissions do not include a separate admin role
- creator-only actions are enforced by backend checks, not by UI visibility alone

## Principle

The creator account is not a family administrator. It is a system-level maintenance account.

Any action that affects more than one family, more than one account, or the app runtime itself belongs in the creator/admin layer.

## What remains member-only

These actions are part of the normal family workflow and are not creator-only:

- parent login
- child add / edit / delete within a family
- family name updates
- parent family account management within the same family
- reward record and approval flows inside the same family
- child score tracking and reward shop access for the current family

## What is creator-only

These actions belong to the developer/operator layer:

- viewing global app statistics
- listing all creator accounts and app owners
- resetting creator state and ensuring one creator remains active
- performing cross-family diagnostics
- platform maintenance and support work
- app recovery or remediation during incidents

## Required enforcement

All creator-only actions must:

1. require a valid login session
2. call `requireAppCreator()` before doing any work
3. validate CSRF on POST requests
4. use family scoping for family-level operations
5. never rely on a hidden UI button as the only protection

## Creator admin page

The creator admin page is the system maintenance surface for the owner.

It should include:

- app overview counts
- creator account listing
- global health summary
- safe maintenance actions
- no family-management operations that are not explicitly allowed by the member flow

## Safe operational rules

### 1. Keep one creator only

The creator state should be intentionally restricted to the application owner account.

When resetting creator state:

- clear other creator flags
- leave the current maintainer account as the only creator
- record the action in logs if logs are added later

### 2. Do not treat creator as a household admin

The creator is not empowered to act as a family admin by default.

Even if the creator belongs to a family, the creator must still use the system-level rules for platform management.

### 3. Use family checks for family actions

Family-only operations must still use a family boundary check, even for creator actions.

This is important because a creator may be a system maintainer, but the application still needs to protect household boundaries.

### 4. Never hide sensitive actions behind client-side logic alone

Frontend controls are only UI hints.

The real enforcement is all in PHP server-side checks.

## Recommended action checklist

Use this when performing creator-level maintenance:

- confirm the current account is the creator
- confirm the target action is truly creator-level
- confirm CSRF token is present
- confirm the action is necessary and reversible
- record what was changed and why
- validate the result immediately after the action

## Example safe creator actions

- reset creator flags
- inspect global family counts
- review creator account mapping
- verify app-wide system health
- perform a maintenance reset after a data issue

## Example actions that must not be done via creator access

- changing a family member's data without family scope
- modifying a child outside the current family boundary
- acting as a household admin without family ownership validation
- bypassing the app's member-level security model for normal family operations

## Final rule

The creator account exists to operate the app, not to override the family model itself.

The family model remains protected by family-scoped checks, while the creator role remains reserved for app-level maintenance and platform integrity.

# Creator-only Permissions Model

## Purpose

This application treats every family parent as a normal member of the household. There is no family-level admin/member role distinction for the end-user experience.

Instead, the application distinguishes between:

- member-level access: normal family parent account
- creator-level access: application owner / system maintainer

This model is intentional. It keeps family members simple while reserving high-risk operations for the creator/maintainer who controls the app itself.

## Core rule

The creator is not a special family administrator. The creator is a system-level superuser.

This means:

- ordinary parents can manage their own family and children within the family boundary
- the creator can perform system-level maintenance that regular members should never access
- the permission check is enforced on the server side, not in the browser UI

## Why this design

A family-level admin/member model is useful for multi-parent households, but it is not the right fit when the goal is: "I, as the app owner, am the only person who can perform platform-level operations."

In this project, the creator is conceptually similar to:

- app owner
- system maintainer
- developer / operator

They are not a household manager in the same way as a normal parent user.

## Security model

### Member permissions

A normal parent user may:

- log in to the parent dashboard
- record points for children in their own family
- manage their household children
- update household settings for their family
- add or remove family parent accounts as allowed by the app flow

A normal member may not:

- access creator-only system pages
- change creator flags
- access global app statistics
- perform maintenance actions across families
- bypass family-level boundary checks

### Creator permissions

A creator may perform:

- system overview checks
- global platform maintenance
- creator-only admin operations
- account repair and cross-family instrumentation
- app-level resets that are unsafe for ordinary family members

The creator permission is enforced by backend checks such as:

- `isAppCreator()`
- `requireAppCreator()`

These functions are the final authority and must not be replaced by UI-only logic.

## Implementation details

The current implementation uses a database flag on the parent account:

- `parent_users.is_creator`

The app owner account is assigned this flag to identify the creator.

In addition, the configuration layer may also define:

- `APP_CREATOR_PARENT_ID`
- `APP_CREATOR_PARENT_IDS`

These are optional safety hooks for code-level configuration and deployment-specific overrides.

The current project enforces the rule in the backend with a function like:

- `requireAppCreator()`

This performs the final access check and exits with an HTTP 403 if the current user is not the creator.

## Frontend vs backend

The frontend may hide or show creator-only controls, but that is only a UX convenience.

The real boundary is backend enforcement.

That means:

- a member user should not see the creator admin page
- even if they manually access the URL, the server must deny access
- any creator-only action must validate permission before performing the database change

## Expected behavior

A normal member flow should look like this:

- login as family parent
- access own family dashboard
- manage own household
- no creator-only features visible

A creator flow should look like this:

- login as creator account
- access creator-only tools
- review app-wide stats
- perform maintenance safely

## Operational guidance

The creator account should remain:

- minimal in number
- intentionally controlled
- protected by backend checks
- not exposed as a family admin role

This keeps the app simple:

- the household model remains family-scoped
- the creator model remains app-scoped

## Summary

The app currently uses a clean separation:

- family members: normal household users
- creator: system owner / runtime maintainer

This is the correct fit for the project because the creator is a developer/operator concern, not a family-management role.

# 当前用户流

## 1. 公开入口

访问站点时，用户先看到 `landing.php`。

这是一个公开页面，不展示任何家庭、孩子、积分或奖励数据。它只承担入口作用，提供两类入口：

- 家长入口：登录到家长系统
- 孩子访问：通过孩子专属访问码或访问链接直接进入该孩子的页面

## 2. 家长登录流

家长从 `parent-login.php` 登录。

登录成功后，进入 `parent-dashboard.php`。

家长控制台负责的核心功能包括：

- 查看本家庭的孩子
- 为每个孩子生成或重置访问码
- 复制孩子访问链接
- 记录积分
- 管理奖励审批
- 进入家庭设置和其他管理页面

## 3. 孩子访问流

家长为某个孩子生成访问链接，例如：

`shop.php?child_id=12&access_code=ABCD1234`

孩子打开这个链接后，可以直接进入自己对应的商店页面。

这个页面只展示：

- 该孩子自己的积分
- 该孩子自己的奖励列表
- 该孩子自己的兑换功能

这个页面不会展示其他家庭或其他孩子的数据。

## 4. 家长登录后家庭视图

已登录的家长仍然可以访问 `index.php`。

但这个页面不再是公共首页，而是“家庭视图/管理视图”的一部分。它用于查看当前家庭中所有孩子的状态，并受 `family_id` 约束，不允许跨家庭访问。

## 5. 设计目标

- 公开入口不泄露家庭内容
- 家长主体需要登录访问家庭管理页
- 孩子只看自己，不需要独立账号
- 每个孩子通过单独访问码/链接绑定自己与家庭
- 后续扩展时，仍然可以在不扩展 child login 账号体系的前提下支持更多家庭和更多孩子

## 6. 关键结论

当前系统的核心用户路径已从“所有孩子首页”转变为：

公开入口 → 家长登录 → 家长控制台 → 单孩子访问页

这更适合多家庭、多孩子和弱认证场景，并使家庭数据隔离更清晰。

# Multi-Family DB Migration Checklist

## Goal

Prepare the database for supporting multiple children per family and multiple families in the same system, while preserving compatibility with the current app behavior.

## Current assumptions to remove

- Some pages and routes are still implicitly tied to a fixed or small set of child records.
- There is no explicit family-level boundary between parent accounts and children.
- Some app logic assumes a single household context instead of a family-scoped model.

## Target data model

### Families table

Create a `families` table with the minimum useful fields:

```sql
CREATE TABLE families (
    id INT UNSIGNED NOT NULL AUTO_INCREMENT,
    name VARCHAR(255) NOT NULL,
    created_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    PRIMARY KEY (id)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
```

### Parent accounts

Add a `family_id` column to the parent-user table (or whichever table stores parent accounts). Example:

```sql
ALTER TABLE parent_users
    ADD COLUMN family_id INT UNSIGNED NULL AFTER id;
```

If the table uses a different name than `parent_users`, replace it with the real table name in the live schema.

### Children

Add a `family_id` column to the children table:

```sql
ALTER TABLE children
    ADD COLUMN family_id INT UNSIGNED NULL AFTER id;
```

### Indexes

```sql
ALTER TABLE parent_users
    ADD INDEX idx_parent_users_family_id (family_id);

ALTER TABLE children
    ADD INDEX idx_children_family_id (family_id);
```

---

## Backfill plan

Before enabling strict rules, backfill all existing rows into a default family or into logical household groups.

### Option A: single default family for legacy records

```sql
INSERT INTO families (name, created_at)
VALUES ('Default Family', NOW());

UPDATE parent_users
SET family_id = 1
WHERE family_id IS NULL;

UPDATE children
SET family_id = 1
WHERE family_id IS NULL;
```

This is the simplest migration if the current app only has one logical family in production.

### Option B: create one family per current parent household

This is more work but better if the app already has a natural grouping of parent/children records in the database.

Example logic:

- For each parent, create a family record
- Assign that family to the parent
- Assign the same family to that parent’s children

This requires a production-safe script and verification before rollout.

---

## Optional foreign keys

If the database can support it and the data is known-good, add foreign keys after backfill:

```sql
ALTER TABLE parent_users
    ADD CONSTRAINT fk_parent_users_family
    FOREIGN KEY (family_id) REFERENCES families(id)
    ON DELETE RESTRICT;

ALTER TABLE children
    ADD CONSTRAINT fk_children_family
    FOREIGN KEY (family_id) REFERENCES families(id)
    ON DELETE RESTRICT;
```

If the project is still in a mixed legacy state, it is safer to add constraints after verifying every row is mapped correctly.

---

## App behavior changes to accompany the schema

### 1. Dynamic child routing

Replace fixed front-end routes like:

- `shop-minecraft.php`
- `shop-music.php`

with a generic route:

- `shop.php?child_id=...`

The page should load the child record by id and apply its theme or settings without hard-coded assumptions.

### 2. Family ownership checks

Use a shared function pattern:

```php
function verifyParentChild(int $parentId, int $childId): bool {
    // ensure parent exists and belongs to same family as child
}
```

And then call it for:

- reward requests
- reward approvals
- point recording
- modification of child records
- parent dashboard filtering

### 3. Family-scoped dashboard queries

Any query used by parent pages should filter by `family_id` as the primary data boundary when listing children, rewards, redemptions, and reports.

---

### Phase 1: Schema addition

- add `families` table
- add `family_id` columns
- add indexes
- backfill legacy rows

### Phase 2: App compatibility

- deploy app code that is safe with `family_id = NULL` and legacy rows
- keep existing UI flow working while new family logic is introduced

- add ownership checks and family-scoped queries

- log violations for review

### Phase 4: Multi-child UI cleanup

- unify theme routing
- add child listing/filtering improvements

1. Back up production DB

5. Run smoke tests for parent access and reward flows

Before declaring the migration complete, confirm:

- reward requests and approvals still work for existing children

# Multi-Child / Multi-Family Migration Plan

## Context: local auth.php vs production auth.php

There are two different purposes for the `auth.php` file depending on environment:

- Local MAMP development: the version in the project root is intended for the local database and service setup on the developer machine.
- Production HostGator deployment: the backup copy of `auth.php` is intentionally configured for the live host environment, with production-safe settings and runtime assumptions.

These are not meant to be treated as the same file with the same credentials or runtime paths. The safe pattern is to keep environment-specific configuration in a shared loader (for example `includes/config.php`) and keep local-only secrets outside the Git repo.

Example rule:

- `includes/config.php` loads values from `includes/config.local.php` or environment variables when present.
- Local secrets remain local-only and are not committed.
- Production credentials live on the live environment and are not shared in source control.

This distinction matters for migration work because the codebase should be environment-agnostic, while the auth settings remain environment-specific.

---

## Architecture audit summary

### What is already strong

- The app is mostly data-driven for child records, rewards, and redemption flows.
- `children`, `rewards`, and `redemptions` are already structured around `child_id`, which is a good base for supporting more children.
- Parent auth is enforced via server-side checks (`includes/auth.php` and route guards), which is the right foundation for adding family-level ownership checks.
- The existing reward flow already treats the child as part of the reward request and the approval path, which supports multi-child logic if routed correctly.

### What is currently blocking full multi-child / multi-family support

1. Hard-coded per-child shop pages
   - `shop-minecraft.php` and `shop-music.php` each assume a fixed child id (`1` and `2`).
   - This is the clearest sign that the app currently expects a small, fixed set of children and does not yet support dynamic routing.

2. Missing family ownership boundary
   - Parents and children need an explicit family relationship so a parent from Family A cannot manage a child in Family B.
   - At present, the code has session auth, but not a clearly centralized family-scope check for all write operations.

3. Theme/route assumptions
   - Some flows still assume a per-theme file or a fixed theme route rather than a generic child-based route.
   - This should be replaced by dynamic page loading based on child id or slug.

4. No explicit multi-family model yet
   - There is no `families` table or `family_id` relationship layer to separate records cleanly by household.

---

## Recommended migration strategy

### Phase 0 — Preparation and safe git setup

Before major function changes, set up a clean Git workflow:

- Initialize or confirm a Git repo.
- Add `.gitignore` entries for local config files and other secret or generated files.
- Add `includes/config.example.php` as the committed template.
- Create a branch strategy:
  - `main` for production-safe/stable code
  - `staging` for validation
  - feature branches for each major change set
- Keep environment secrets out of Git completely.

This is the safest starting point because it makes the app changes reviewable and reversible.

### Phase 1 — Add family model in the database

Add a family layer before reworking the app logic:

- `families` table
  - `id`
  - `name`
  - `created_at`
- Add `family_id` to `parents`
- Add `family_id` to `children`
- Backfill existing records into a default family or the active household grouping
- Add indexes on `children.family_id` and `parents.family_id`

Optional but recommended:

- Add `parent_role` or `family_role` later if different guardian roles are needed
- Add a per-family settings table if the app grows beyond a basic household model

### Phase 2 — Replace hard-coded child pages with dynamic routing

Replace the fixed shop pages with a single dynamic route, for example:

- `shop.php?child_id=123`
- or `shop.php?child_slug=alex`

Implementation goals:

- Resolve the child from database
- Load that child’s theme (`minecraft`, `music`, etc.)
- Load its rewards and redemption status
- Keep the route generic instead of hard-coded to one child or one theme

If the current site needs a simple interim step, a compatibility layer can redirect older routes to the correct child id while the app moves to a single dynamic flow.

### Phase 3 — Centralize ownership and parent access checks

Add a shared function such as:

- `verify_parent_child(parent_id, child_id)`
- `verify_parent_family(parent_id, family_id)`

Use this in every write endpoint:

- reward request
- reward approval
- point recording
- child management actions
- parent dashboard filters

This prevents one family from acting on another family’s child records.

### Phase 4 — Validate UI and reporting at scale

After the logic is dynamic and safe, update UI flows:

- child list on the home page is already data-driven and should work naturally
- reward pages should not assume a single static theme page
- parent dashboard should show per-family and per-child information clearly
- reports should aggregate by child and by family as needed

### Phase 5 — Production rollout and monitoring

Roll out only after:

- staging validation
- DB migration testing
- backup/recovery validation
- smoke tests on the live environment using non-production data or a staging snapshot

Then deploy the app update and monitor:

- parent access errors
- child reward flows
- duplicate reward submissions
- auth/session issues

---

## Recommended sequencing: app changes vs GitHub setup

The safest order is:

1. Set up Git repo and branch workflow first
   - repo structure
   - `.gitignore`
   - `config.example.php`
   - safe backup of local env files

2. Create a staging branch and a migration branch for the major changes
   - develop/staging branch for pre-production validation
   - feature branch for the multi-family work

3. Test the DB schema changes in staging or local copy first
   - no production writes until validated

4. Make the application code changes in the feature branch
   - dynamic child routing
   - family ownership checks
   - UI updates

5. Validate with a smoke test suite and manual checks
   - parent/child isolation
   - shop rendering by child
   - approval flows
   - permissions and redirects

6. Merge into staging, validate, then merge to a production release branch

7. Deploy to HostGator only after final checks

This order keeps the codebase reviewable and reduces the risk that large logic changes and environment setup get tangled together.

### Why this order is better

- It keeps credentials and environment config out of source control.
- It allows the team to test the migration on a safe branch before touching production.
- It prevents large app rewrites from being interwoven with repo setup problems.
- It gives a clean audit trail if something breaks during the migration.

In short: do the Git hygiene and branch structure first, then do the major app migration on a dedicated change branch, and only deploy after validating the schema and behavior on a staging copy.

---

## Suggested practical workflow for this project

For this codebase, I would use this sequence:

1. Confirm git repo status and create `.gitignore` / environment-template files
2. Create a dedicated `feature/multi-family` branch
3. Add the database migration scripts for `families` and `family_id`
4. Run the migration in a local or staging database
5. Update the app to be child-driven and family-aware
6. Validate the reward and approval flows
7. Merge to staging
8. Test with a nearly production-like dataset
9. Deploy to live only after the release check passes

This gives a clean path from the current app to the future multi-family architecture without losing the existing app behavior during the transition.

## Short recommendation

If the team wants a safe, low-drama migration, I recommend:

- keep the environment-specific `auth.php` files separate by deployment target
- keep production credentials out of Git
- use a small, explicit migration plan like the one above
- do the Git setup and branch discipline before the major app refactor

That approach gives both operational safety and a future-proof path to multi-child and multi-family support.

const ACTIVE_WINDOWS = [
  { start: "00:00", end: "23:59", interval: 5000 }
];

let pollTimer = null;
let lastVersion = null;
let retryDelay = 10000;

function getActiveInterval() {
  const now = new Date();
  const minutes = now.getHours() * 60 + now.getMinutes();

  for (const window of ACTIVE_WINDOWS) {
    const [startHour, startMinute] = window.start.split(":").map(Number);
    const [endHour, endMinute] = window.end.split(":").map(Number);

    const start = startHour * 60 + startMinute;
    const end = endHour * 60 + endMinute;

    if (minutes >= start && minutes < end) {
      return window.interval;
    }
  }

  return null;
}

async function fetchJson(url) {
  const response = await fetch(url, { cache: "no-store" });

  if (!response.ok) {
    throw new Error("Network response was not ok");
  }

  return response.json();
}

function renderChildren(children) {
  children.forEach(child => {
    const card = document.querySelector(`[data-child-id="${child.id}"]`);

    if (!card) return;

    const balance = card.querySelector("[data-balance]");
    const daily = card.querySelector("[data-daily]");
    const dailyBar = card.querySelector("[data-daily-bar]");
    const pool = card.querySelector("[data-pool]");
    const poolBar = card.querySelector("[data-pool-bar]");

    const oldBalance = Number(balance?.textContent || 0);
    const newBalance = Number(child.normal_balance);

    if (balance) balance.textContent = newBalance;

    if (daily) {
      daily.textContent = `${child.daily_points_today} / ${child.daily_cap}`;
    }

    if (dailyBar) {
      dailyBar.style.width =
        Math.min(100, child.daily_points_today / child.daily_cap * 100) + "%";
    }

    if (pool) {
      pool.textContent = `${child.quarterly_pool} / ${child.quarterly_goal}`;
    }

    if (poolBar) {
      poolBar.style.width =
        Math.min(100, child.quarterly_pool / child.quarterly_goal * 100) + "%";
    }

    if (newBalance > oldBalance && balance) {
      animateIncrease(balance);
    }
  });
}

function animateIncrease(element) {
  element.animate(
    [
      { transform: "scale(1)", opacity: 1 },
      { transform: "scale(1.25)", opacity: 1 },
      { transform: "scale(1)", opacity: 1 }
    ],
    {
      duration: 700,
      easing: "ease-out"
    }
  );
}

async function checkForUpdates() {
  try {
    const versionData = await fetchJson("api/version.php");

    if (lastVersion === null) {
      lastVersion = versionData.v;
      const statusData = await fetchJson("api/status.php");
      renderChildren(statusData.children);
      return;
    }

    if (versionData.v !== lastVersion) {
      lastVersion = versionData.v;
      const statusData = await fetchJson("api/status.php");
      renderChildren(statusData.children);
    }

    retryDelay = 10000;
  } catch (error) {
    retryDelay = Math.min(retryDelay * 2, 60000);
  }
}

function schedulePolling() {
  clearTimeout(pollTimer);

  if (document.hidden) return;

  const interval = getActiveInterval();

  if (!interval) return;

  pollTimer = setTimeout(async () => {
    await checkForUpdates();
    schedulePolling();
  }, interval);
}

document.addEventListener("visibilitychange", () => {
  if (!document.hidden) {
    checkForUpdates();
  }

  schedulePolling();
});

checkForUpdates();
schedulePolling();
